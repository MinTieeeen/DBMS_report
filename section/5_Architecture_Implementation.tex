\subsection{Thiết kế kiến trúc hệ thống}
\label{subsec:architecture}

\subsubsection{Kiến trúc tổng thể}
\label{subsubsec:architecture-overview}

Novelle được thiết kế theo kiến trúc client--server nhiều tầng (\emph{n-tier}). Người dùng thao tác trên ứng dụng web chạy trong trình duyệt; ứng dụng này giao tiếp với backend qua REST API dùng định dạng JSON; backend xử lý nghiệp vụ và làm việc với hai hệ quản trị cơ sở dữ liệu là MS SQL Server và Redis. Hình~\ref{fig:component-diagram} mô tả các thành phần chính của hệ thống và cách chúng giao tiếp với nhau.

\begin{figure}[H]
    \centering
    \includegraphics[width=\textwidth]{image/chapter5/component-diagram.png}
    \caption{Sơ đồ thành phần (component diagram) của hệ thống Novelle}
    \label{fig:component-diagram}
\end{figure}

\paragraph{Frontend.} Ứng dụng web được xây dựng dưới dạng SPA (\emph{Single Page Application}): trình duyệt chỉ tải ứng dụng một lần, sau đó mọi thao tác đều được xử lý phía client và chỉ gọi backend để lấy hoặc gửi dữ liệu. Frontend gồm ba thành phần:
\begin{itemize}
    \item \textbf{UI Pages \& Components:} các trang và thành phần giao diện cho người đọc, khu vực sáng tác và khu vực quản trị.
    \item \textbf{Server State (TanStack Query):} quản lý dữ liệu lấy từ server, lưu tạm kết quả và tự làm mới khi dữ liệu thay đổi.
    \item \textbf{API Client (Axios):} nơi duy nhất gửi yêu cầu HTTP tới backend, kèm cookie phiên đăng nhập và mã chống CSRF.
\end{itemize}

\paragraph{Backend.} Backend là một ứng dụng Spring Boot, được tổ chức theo các lớp rõ ràng; mỗi yêu cầu đi lần lượt qua từng lớp:
\begin{itemize}
    \item \textbf{Security Filter Chain:} kiểm tra phiên đăng nhập, trạng thái tài khoản và quyền truy cập theo vai trò trước khi yêu cầu tới được lớp xử lý.
    \item \textbf{REST Controllers:} tiếp nhận yêu cầu, kiểm tra dữ liệu đầu vào và trả kết quả dạng JSON.
    \item \textbf{Services:} chứa logic nghiệp vụ và quản lý giao dịch; quyết định dữ liệu nào đọc từ Redis, dữ liệu nào đọc hoặc ghi xuống SQL Server.
    \item \textbf{MyBatis Mappers:} thực thi các câu lệnh SQL, thủ tục lưu trữ trên SQL Server và ánh xạ kết quả sang đối tượng Java.
    \item \textbf{Redis Infrastructure:} tập hợp các thành phần làm việc với Redis gồm bộ nhớ đệm, lưu phiên đăng nhập, bộ đếm lượt xem, bản nháp, giới hạn tần suất và khoá phân tán.
    \item \textbf{Scheduled Jobs:} các tác vụ chạy định kỳ, như đồng bộ lượt xem từ Redis xuống SQL Server và tự động đăng các chương đã hẹn giờ.
\end{itemize}

\paragraph{Tầng dữ liệu.} Hai hệ quản trị cơ sở dữ liệu được dùng với vai trò khác nhau:
\begin{itemize}
    \item \textbf{MS SQL Server} là nơi lưu trữ chính cho toàn bộ dữ liệu nghiệp vụ; các ràng buộc toàn vẹn, giao dịch, trigger và thủ tục lưu trữ đều nằm ở đây.
    \item \textbf{Redis} là lớp dữ liệu bổ trợ trong bộ nhớ, giữ những dữ liệu cần truy xuất nhanh hoặc chỉ có giá trị tạm thời. Mất dữ liệu trên Redis không làm hỏng dữ liệu nghiệp vụ trên SQL Server.
\end{itemize}

\paragraph{Luồng xử lý một yêu cầu.} Khi người dùng mở trang chi tiết một truyện, giao diện gửi yêu cầu qua API Client tới backend. Security Filter Chain xác định người dùng từ phiên lưu trên Redis, sau đó Controller chuyển yêu cầu cho Service tương ứng. Service tìm dữ liệu trong bộ nhớ đệm trên Redis trước; nếu chưa có, Service gọi Mapper truy vấn SQL Server, rồi lưu kết quả vào Redis cho các lần đọc sau. Với các thao tác ghi như đánh giá hay đăng chương, Service ghi xuống SQL Server trong một giao dịch, sau đó xoá các khoá đệm liên quan trên Redis để dữ liệu hiển thị luôn được cập nhật.

\subsubsection{Sơ đồ gói (Package diagram)}
\label{subsubsec:package-diagram}

Mã nguồn của Novelle được tổ chức thành hai dự án: \texttt{novelle\_frontend} và \texttt{novelle\_backend}. Mỗi dự án được chia thành các gói (package) theo lớp kỹ thuật; quan hệ phụ thuộc giữa các gói được thể hiện ở Hình~\ref{fig:package-diagram}, trong đó mũi tên nét đứt đi từ gói sử dụng tới gói được sử dụng.

\begin{figure}[H]
    \centering
    \includegraphics[width=\textwidth]{image/chapter5/package-diagram.png}
    \caption{Sơ đồ gói của hệ thống Novelle}
    \label{fig:package-diagram}
\end{figure}

\paragraph{Các gói của frontend.}
\begin{itemize}
    \item \texttt{app}: điểm khởi tạo ứng dụng, khai báo bộ định tuyến (router) và các provider dùng chung.
    \item \texttt{pages}: các trang ứng với từng đường dẫn, chia theo khu vực người đọc, tài khoản, khu vực sáng tác và khu vực quản trị.
    \item \texttt{components}: các thành phần giao diện tái sử dụng, nhóm theo chức năng (truyện, đọc chương, thư viện, sáng tác, quản trị) và bộ thành phần giao diện cơ bản.
    \item \texttt{layouts}: khung bố cục chung của từng khu vực (thanh điều hướng, thanh bên, chân trang).
    \item \texttt{hooks}: các hook React đóng gói việc gọi API, quản lý trạng thái và xử lý thao tác của người dùng.
    \item \texttt{api}: các hàm gọi REST API của backend, nhóm theo module nghiệp vụ, dùng chung một HTTP client.
    \item \texttt{auth}: quản lý trạng thái đăng nhập và chặn truy cập các trang theo vai trò.
    \item \texttt{types}, \texttt{lib}, \texttt{utils}: kiểu dữ liệu TypeScript dùng chung và các hàm tiện ích.
\end{itemize}

\paragraph{Các gói của backend.}
\begin{itemize}
    \item \texttt{controller}: các REST controller nhận yêu cầu và trả kết quả JSON.
    \item \texttt{service}: logic nghiệp vụ và quản lý giao dịch.
    \item \texttt{mapper}: các MyBatis mapper chứa câu lệnh SQL và lời gọi thủ tục lưu trữ trên SQL Server.
    \item \texttt{model}: các đối tượng nhận kết quả truy vấn từ cơ sở dữ liệu.
    \item \texttt{dto}: các đối tượng dữ liệu vào/ra của API kèm quy tắc kiểm tra hợp lệ.
    \item \texttt{exception}: mã lỗi, ngoại lệ và bộ xử lý lỗi tập trung.
    \item \texttt{config}: cấu hình bảo mật, CORS, lịch chạy tác vụ và tài liệu API.
    \item \texttt{infrastructure}: các thành phần hạ tầng dùng chung, gồm \texttt{security} (phiên đăng nhập, kiểm tra trạng thái tài khoản), \texttt{ratelimit} (giới hạn tần suất), \texttt{cache} (bộ nhớ đệm, bản nháp, khoá phân tán trên Redis), \texttt{views} (đếm và đồng bộ lượt xem), \texttt{jobs} (tác vụ đăng chương hẹn giờ), \texttt{html} (làm sạch HTML, đếm chữ, sinh slug) và \texttt{mybatis} (bộ chuyển đổi kiểu dữ liệu đặc thù của SQL Server).
\end{itemize}

Các gói phụ thuộc một chiều theo lớp: \texttt{controller} $\rightarrow$ \texttt{service} $\rightarrow$ \texttt{mapper} $\rightarrow$ \texttt{model}. Gói \texttt{api} của frontend là nơi duy nhất giao tiếp với backend, thông qua các endpoint do \texttt{controller} cung cấp.

\paragraph{Module Nền tảng.} Cung cấp các cơ chế dùng chung cho mọi module khác: lưu phiên đăng nhập trên Redis, kiểm tra trạng thái tài khoản ở mỗi yêu cầu, phân quyền theo vai trò, giới hạn tần suất các thao tác nhạy cảm, mô hình lỗi thống nhất cho toàn bộ API và bộ nhớ đệm dùng chung. Ở frontend, module này gồm HTTP client, cơ chế lưu trữ trạng thái dữ liệu từ server và bộ chặn truy cập theo vai trò.

\paragraph{Module Tài khoản.} Chịu trách nhiệm đăng ký tài khoản (mặc định là người đọc), đăng nhập bằng tên đăng nhập hoặc email, đăng xuất, xem và sửa hồ sơ cá nhân, đổi mật khẩu và hiển thị hồ sơ công khai. Mật khẩu được băm bằng BCrypt; việc sửa hồ sơ dùng khoá lạc quan để phát hiện chỉnh sửa đồng thời. Khi vai trò của người dùng thay đổi (được duyệt làm tác giả) hoặc tài khoản bị khoá, thay đổi có hiệu lực ngay ở yêu cầu kế tiếp.

\paragraph{Module Danh mục \& truyện.} Phục vụ toàn bộ phần đọc công khai: trang chủ với các danh sách truyện, danh sách thể loại và gợi ý thẻ, tìm kiếm toàn văn kết hợp bộ lọc nhiều điều kiện, trang chi tiết truyện kèm mục lục theo tập, và trang tác giả. Module này chỉ đọc dữ liệu, luôn ẩn truyện chưa công khai và chương chưa đăng, và dùng bộ nhớ đệm Redis cho trang chủ, danh sách thể loại và chi tiết truyện.

\paragraph{Module Đọc truyện.} Trả nội dung chương cùng liên kết chương trước và chương sau, ghi nhận lượt xem và lưu vị trí đọc của người dùng. Lượt xem được đếm trên Redis, chống tính trùng trong 30 phút, rồi được tác vụ định kỳ đồng bộ xuống SQL Server mỗi 60 giây. Nội dung chương đã đăng được lưu đệm trên Redis.

\paragraph{Module Thư viện \& đánh giá.} Quản lý các hoạt động cá nhân của người đọc: lịch sử đọc, theo dõi truyện và tác giả, tủ truyện (tạo, sửa, xoá, thêm truyện, sắp xếp thứ tự) và đánh giá truyện. Các số liệu tổng hợp như lượt theo dõi và điểm trung bình được trigger trên SQL Server cập nhật; sau mỗi thao tác ghi, module xoá bộ nhớ đệm của truyện liên quan.

\paragraph{Module Yêu cầu tác giả.} Cho phép người đọc gửi, theo dõi và huỷ yêu cầu trở thành tác giả, và cho phép quản trị viên xem hàng đợi, duyệt hoặc từ chối yêu cầu. Việc duyệt được thực hiện trong một thủ tục lưu trữ: đổi trạng thái yêu cầu, đổi vai trò tài khoản và tạo hồ sơ tác giả trong cùng một giao dịch, đồng thời bảo đảm mỗi yêu cầu chỉ được xử lý một lần kể cả khi hai quản trị viên thao tác cùng lúc.

\paragraph{Module Studio sáng tác.} Khu vực làm việc của tác giả, gồm ba phần:
\begin{itemize}
    \item \textbf{Quản lý truyện:} tạo, sửa, công khai, chuyển về bản nháp, xoá truyện; gán thể loại, thẻ; cập nhật hồ sơ tác giả.
    \item \textbf{Viết chương:} quản lý tập và chương, tự động lưu bản nháp lên Redis, lưu chương với khoá lạc quan, đăng ngay hoặc hẹn giờ đăng, xem và khôi phục các phiên bản cũ.
    \item \textbf{Thống kê tác giả:} số liệu tổng quan, lượt xem theo ngày hoặc tháng, thống kê từng truyện và người theo dõi.
\end{itemize}
Mọi thao tác chỉ áp dụng trên truyện của chính tác giả đang đăng nhập.

\paragraph{Module Quản trị.} Dành cho quản trị viên, gồm quản lý người dùng (tìm kiếm, khoá, mở khoá, tạo quản trị viên, cấp dấu xác minh), kiểm duyệt nội dung (ẩn, hiện, xoá truyện; xoá nhận xét vi phạm), quản lý thể loại và thẻ, và thống kê toàn hệ thống. Các truy vấn thống kê dùng nhiều phép tổng hợp và truy vấn con, kết quả được lưu đệm ngắn hạn trên Redis.

\subsubsection{Module Danh mục \& truyện}
\label{subsubsec:feature-catalog}

\paragraph{Mô tả chức năng.} Module phục vụ toàn bộ phần khám phá truyện công khai, gồm các yêu cầu chức năng nhóm FR-2 và FR-3:
\begin{itemize}
    \item \textbf{Trang chủ:} năm danh sách truyện, mỗi danh sách tối đa 12 truyện: mới cập nhật, mới đăng, đánh giá cao (chỉ xét truyện có từ 10 lượt đánh giá), xem nhiều và đã hoàn thành.
    \item \textbf{Danh mục:} danh sách thể loại kèm số truyện công khai của từng thể loại, và gợi ý thẻ theo tiền tố khi người dùng gõ.
    \item \textbf{Tìm kiếm và lọc:} tìm toàn văn theo tên và phần giới thiệu (không phân biệt dấu thanh, hoa thường), kết hợp tự do với các bộ lọc thể loại, thẻ, tình trạng, giới hạn độ tuổi, khoảng số chương, điểm tối thiểu; sắp xếp theo thời gian cập nhật, thời gian đăng, lượt xem, điểm, lượt theo dõi hoặc mức độ liên quan; phân trang.
    \item \textbf{Chi tiết truyện:} thông tin truyện, tác giả, thể loại (thể loại chính đứng đầu), thẻ và mục lục các chương đã đăng, nhóm theo tập.
    \item \textbf{Trang tác giả:} hồ sơ công khai, thống kê tổng hợp trên các truyện đã công khai và danh sách truyện của tác giả.
\end{itemize}

Trang chủ, danh sách thể loại và chi tiết truyện được lưu đệm trên Redis lần lượt trong 2 phút, 1 giờ và 10 phút; chức năng tìm kiếm bị giới hạn 60 lần mỗi phút trên một địa chỉ IP.

\paragraph{Các câu truy vấn SQL.}

\textit{Danh sách thể loại kèm số truyện công khai} (truy vấn kết nối ngoài và hàm tổng hợp):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT g.GenreId, g.Name, g.Slug, g.Description,
       COUNT(s.StoryId) AS StoryCount
FROM dbo.Genre g
LEFT JOIN dbo.StoryGenre sg    ON sg.GenreId = g.GenreId
LEFT JOIN dbo.vw_PublicStory s ON s.StoryId  = sg.StoryId
GROUP BY g.GenreId, g.Name, g.Slug, g.Description, g.SortOrder
ORDER BY g.SortOrder, g.GenreId;
\end{minted}

\textit{Một khối trang chủ: truyện đã hoàn thành, mới cập nhật trước} (truy vấn một điều kiện):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT TOP (12) s.StoryId, s.Slug, s.Title, LEFT(s.Description, 300) AS Description,
       s.CoverUrl, s.CompletionStatus, s.ChapterCount, s.ViewCount, s.RatingAvg,
       s.LastChapterAt, s.AuthorId, s.AuthorName,
       g.GenreId, g.Name AS GenreName, g.Slug AS GenreSlug
FROM dbo.vw_PublicStory s
LEFT JOIN dbo.StoryGenre pg ON pg.StoryId = s.StoryId AND pg.IsPrimary = 1
LEFT JOIN dbo.Genre g       ON g.GenreId  = pg.GenreId
WHERE s.CompletionStatus = 'COMPLETED'
ORDER BY s.LastChapterAt DESC, s.StoryId DESC;
\end{minted}

\textit{Tìm kiếm với từ khoá và nhiều bộ lọc} (truy vấn nhiều điều kiện, truy vấn con, tìm kiếm toàn văn). Mỗi từ khoá được tìm bằng một \texttt{CONTAINSTABLE} riêng; truyện phải khớp đủ mọi từ khoá và đủ mọi thể loại được chọn. Ví dụ với hai từ khoá và hai thể loại:
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT s.StoryId, s.Slug, s.Title, s.CoverUrl, s.ChapterCount, s.RatingAvg,
       s.LastChapterAt, s.AuthorName
FROM dbo.vw_PublicStory s
JOIN (
    SELECT m.[KEY], SUM(m.[RANK]) AS [RANK]
    FROM (
        SELECT [KEY], [RANK], 0 AS TermNo
        FROM CONTAINSTABLE(dbo.Story, (Title, Description), @term1)
        UNION ALL
        SELECT [KEY], [RANK], 1 AS TermNo
        FROM CONTAINSTABLE(dbo.Story, (Title, Description), @term2)
    ) m
    GROUP BY m.[KEY]
    HAVING COUNT(DISTINCT m.TermNo) = 2
) ft ON ft.[KEY] = s.StoryId
WHERE s.CompletionStatus = @completionStatus
  AND s.ChapterCount >= @minChapters
  AND s.RatingAvg    >= @minRating
  AND s.StoryId IN (
        SELECT sg.StoryId
        FROM dbo.StoryGenre sg
        JOIN dbo.Genre gg ON gg.GenreId = sg.GenreId
        WHERE gg.Slug IN (@genre1, @genre2)
        GROUP BY sg.StoryId
        HAVING COUNT(DISTINCT sg.GenreId) = 2)
ORDER BY ft.[RANK] DESC, s.StoryId DESC
OFFSET @offset ROWS FETCH NEXT @size ROWS ONLY;
\end{minted}
Câu đếm tổng số kết quả dùng cùng phần \texttt{FROM}, \texttt{JOIN} và \texttt{WHERE}, chỉ thay danh sách cột bằng \texttt{COUNT(*)} và bỏ phần sắp xếp, phân trang.

\textit{Chi tiết truyện kèm tác giả} (truy vấn kết nối):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT s.StoryId, s.Slug, s.Title, s.Description, s.CoverUrl, s.CompletionStatus,
       s.AgeRating, s.ChapterCount, s.WordCount, s.ViewCount, s.FollowCount,
       s.RatingAvg, s.RatingCount, s.LastChapterAt, s.PublishedAt,
       a.UserId AS AuthorId, a.PenName, a.IsVerified
FROM dbo.vw_PublicStory s
JOIN dbo.Author a ON a.UserId = s.AuthorId
WHERE s.StoryId = @storyId;
\end{minted}

\textit{Mục lục theo tập} (truy vấn kết nối nhiều điều kiện, chỉ lấy chương đã đăng):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT v.VolumeId, v.VolumeNo, v.Title AS VolumeTitle,
       ch.ChapterId, ch.ChapterNo, ch.Title AS ChapterTitle, ch.PublishedAt
FROM dbo.Volume v
JOIN dbo.Chapter ch ON ch.VolumeId  = v.VolumeId
                   AND ch.StoryId   = v.StoryId
                   AND ch.Status    = 'PUBLISHED'
                   AND ch.IsDeleted = 0
WHERE v.StoryId = @storyId
ORDER BY v.VolumeNo, ch.ChapterNo;
\end{minted}

\textit{Hồ sơ tác giả kèm thống kê trên các truyện đã công khai} (truy vấn con với \texttt{CROSS APPLY} và hàm tổng hợp; điểm trung bình được tính có trọng số theo số lượt đánh giá):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT a.UserId AS AuthorId, a.PenName, a.Bio, u.AvatarUrl, a.IsVerified,
       a.FollowerCount, a.CreatedAt AS AuthorSince,
       st.StoryCount, st.ChapterCount, st.ViewCount, st.RatingCount, st.RatingAvg
FROM dbo.Author a
JOIN dbo.[User] u ON u.UserId = a.UserId
CROSS APPLY (
    SELECT COUNT(*)                                         AS StoryCount,
           COALESCE(SUM(CAST(p.ChapterCount AS BIGINT)), 0) AS ChapterCount,
           COALESCE(SUM(p.ViewCount), 0)                    AS ViewCount,
           COALESCE(SUM(CAST(p.RatingCount AS BIGINT)), 0)  AS RatingCount,
           COALESCE(CAST(SUM(CAST(p.RatingAvg AS DECIMAL(18,4)) * p.RatingCount)
                         / NULLIF(SUM(CAST(p.RatingCount AS DECIMAL(18,4))), 0)
                    AS DECIMAL(3,2)), 0)                    AS RatingAvg
    FROM dbo.vw_PublicStory p
    WHERE p.AuthorId = a.UserId
) st
WHERE a.UserId = @authorId;
\end{minted}

\subsubsection{Module Đọc truyện}
\label{subsubsec:feature-reading}

\paragraph{Mô tả chức năng.} Module hiện thực nhóm yêu cầu FR-4:
\begin{itemize}
    \item \textbf{Đọc chương:} trả về tên chương, nội dung, số chữ, ngày đăng, lượt xem, thông tin truyện, tập (nếu có) và mã của chương liền trước, liền sau trong số các chương đã đăng. Chương nháp, chương hẹn giờ, chương đã xoá hoặc thuộc truyện không công khai đều trả về cùng một lỗi \texttt{404}.
    \item \textbf{Ghi nhận lượt xem:} mỗi lần đọc thành công tăng một lượt xem cho chương và cho truyện theo ngày (giờ Việt Nam). Một người xem (theo tài khoản, hoặc theo địa chỉ IP nếu chưa đăng nhập) chỉ được tính một lượt cho mỗi chương trong 30 phút.
    \item \textbf{Đồng bộ lượt xem:} lượt xem được đếm trên Redis; mỗi 60 giây, một tác vụ định kỳ gom các bộ đếm và cộng dồn xuống SQL Server trong một giao dịch.
    \item \textbf{Lưu vị trí đọc:} khi thành viên mở chương, hệ thống tự ghi lại chương đang đọc; trong lúc đọc, giao diện gửi thêm vị trí cuộn (0--100\%). Mỗi người chỉ có một vị trí đọc cho mỗi truyện, và chương được lưu phải thuộc đúng truyện đó.
    \item \textbf{Đọc tiếp:} trả về chương và vị trí cuộn đã lưu; nếu chương đó không còn công khai thì coi như chưa có vị trí đọc.
\end{itemize}
Nội dung chương (cột \texttt{NVARCHAR(MAX)}, là phần dữ liệu lớn nhất) được lưu đệm trên Redis trong 30 phút, còn thông tin điều hướng và trạng thái công khai luôn được đọc trực tiếp từ SQL Server. Chức năng lưu vị trí đọc bị giới hạn 120 lần mỗi phút trên một tài khoản.

\paragraph{Các câu truy vấn SQL.}

\textit{Thông tin chương kèm chương liền trước, liền sau} (truy vấn kết nối, nhiều điều kiện và hai truy vấn con tương quan):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT ch.ChapterId, ch.ChapterNo, ch.ViewCount, ch.StoryId,
       s.Slug AS StorySlug, s.Title AS StoryTitle, s.AuthorId, s.AuthorName,
       v.VolumeId, v.VolumeNo, v.Title AS VolumeTitle,
       (SELECT TOP (1) p.ChapterId
        FROM dbo.Chapter p
        WHERE p.StoryId = ch.StoryId AND p.ChapterNo < ch.ChapterNo
          AND p.Status = 'PUBLISHED' AND p.IsDeleted = 0
        ORDER BY p.ChapterNo DESC) AS PrevChapterId,
       (SELECT TOP (1) n.ChapterId
        FROM dbo.Chapter n
        WHERE n.StoryId = ch.StoryId AND n.ChapterNo > ch.ChapterNo
          AND n.Status = 'PUBLISHED' AND n.IsDeleted = 0
        ORDER BY n.ChapterNo ASC) AS NextChapterId
FROM dbo.Chapter ch
JOIN dbo.vw_PublicStory s ON s.StoryId  = ch.StoryId
LEFT JOIN dbo.Volume v    ON v.VolumeId = ch.VolumeId
WHERE ch.ChapterId = @chapterId
  AND ch.Status = 'PUBLISHED'
  AND ch.IsDeleted = 0;
\end{minted}

\textit{Nội dung chương}, chỉ đọc khi bộ nhớ đệm chưa có (truy vấn một điều kiện):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT Title, Content, WordCount, PublishedAt
FROM dbo.Chapter
WHERE ChapterId = @chapterId;
\end{minted}

\textit{Lưu vị trí đọc} (thêm mới hoặc cập nhật trong một câu lệnh). Gợi ý \texttt{HOLDLOCK} giữ khoá trên khoảng khoá từ lúc tìm tới lúc ghi, để hai tab cùng mở truyện lần đầu không cùng chèn một dòng:
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
MERGE dbo.ReadingHistory WITH (HOLDLOCK) AS t
USING (SELECT @userId AS UserId, @storyId AS StoryId, @chapterId AS ChapterId,
              CAST(@scrollPercent AS DECIMAL(5,2)) AS Scroll) AS src
   ON t.UserId = src.UserId AND t.StoryId = src.StoryId
WHEN MATCHED THEN
    UPDATE SET LastChapterId = src.ChapterId,
               ScrollPercent = src.Scroll,
               UpdatedAt     = SYSUTCDATETIME()
WHEN NOT MATCHED THEN
    INSERT (UserId, StoryId, LastChapterId, ScrollPercent)
    VALUES (src.UserId, src.StoryId, src.ChapterId, src.Scroll);
\end{minted}

\textit{Vị trí đọc tiếp}, chỉ khi chương đã lưu còn công khai (truy vấn kết nối):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT h.StoryId, h.LastChapterId AS ChapterId, ch.ChapterNo,
       ch.Title AS ChapterTitle, h.ScrollPercent, h.UpdatedAt
FROM dbo.ReadingHistory h
JOIN dbo.Chapter ch ON ch.ChapterId = h.LastChapterId
                   AND ch.Status = 'PUBLISHED' AND ch.IsDeleted = 0
WHERE h.UserId = @userId AND h.StoryId = @storyId;
\end{minted}

\textit{Đồng bộ lượt xem} trong thủ tục \texttt{sp\_FlushViews}. Danh sách lượt xem từ Redis được truyền vào dưới dạng tham số bảng \texttt{@Deltas}; các câu lệnh dưới đây chạy trong cùng một giao dịch (cập nhật có kết nối, hàm tổng hợp và \texttt{MERGE}):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
UPDATE s
SET ViewCount = s.ViewCount + d.Total
FROM dbo.Story s
JOIN (SELECT TargetId, SUM(CAST(Delta AS BIGINT)) AS Total
      FROM @Deltas WHERE TargetType = 'S'
      GROUP BY TargetId) d ON d.TargetId = s.StoryId;

-- câu UPDATE tương tự cho dbo.Chapter với TargetType = 'C'

MERGE dbo.StoryViewDaily WITH (HOLDLOCK) AS t
USING (SELECT s.StoryId, d.ViewDate, SUM(d.Delta) AS Delta
       FROM @Deltas d
       JOIN dbo.Story s ON s.StoryId = d.TargetId
       WHERE d.TargetType = 'S'
       GROUP BY s.StoryId, d.ViewDate) AS src
   ON t.StoryId = src.StoryId AND t.ViewDate = src.ViewDate
WHEN MATCHED THEN
    UPDATE SET Views = t.Views + src.Delta
WHEN NOT MATCHED THEN
    INSERT (StoryId, ViewDate, Views) VALUES (src.StoryId, src.ViewDate, src.Delta);
\end{minted}

\subsubsection{Module Thư viện \& đánh giá}
\label{subsubsec:feature-library}

\paragraph{Mô tả chức năng.} Module hiện thực nhóm yêu cầu FR-5 và FR-6, gồm các hoạt động cá nhân của người đọc:
\begin{itemize}
    \item \textbf{Lịch sử đọc:} danh sách truyện đã đọc, mới đọc trước, kèm chương đọc gần nhất và vị trí cuộn; người dùng có thể xoá từng truyện khỏi lịch sử.
    \item \textbf{Theo dõi truyện và tác giả:} theo dõi hoặc bỏ theo dõi, xem trạng thái theo dõi, xem danh sách truyện đang theo dõi (sắp theo chương mới nhất) và tác giả đang theo dõi. Tác giả không được tự theo dõi chính mình.
    \item \textbf{Tủ truyện:} tạo, sửa, xoá tủ (tối đa 50 tủ, tên không trùng trong các tủ của cùng một người); thêm truyện kèm ghi chú, bỏ truyện, sắp xếp lại thứ tự (tối đa 500 truyện mỗi tủ); tủ công khai được người khác xem, tủ riêng tư chỉ chủ sở hữu xem được.
    \item \textbf{Đánh giá:} chấm điểm 1--5 kèm nhận xét, sửa hoặc xoá đánh giá của mình; mỗi người có tối đa một đánh giá cho mỗi truyện. Quản trị viên và tác giả của chính truyện đó không được đánh giá. Trang truyện hiển thị danh sách nhận xét, điểm trung bình và số lượt đánh giá ở từng mức điểm.
\end{itemize}
Các thao tác theo dõi và đánh giá đều có thể bị gọi lặp lại hoặc gửi đồng thời (bấm nhiều lần, mở nhiều tab), nên các câu lệnh ghi được thiết kế để chạy lặp lại vẫn cho cùng kết quả. \textit{Theo dõi truyện, chỉ chèn khi chưa theo dõi} (thêm dữ liệu với truy vấn con \texttt{NOT EXISTS}; \texttt{UPDLOCK, HOLDLOCK} giữ khoá từ lúc kiểm tra tới lúc chèn):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
INSERT dbo.Follow (UserId, StoryId)
SELECT @userId, @storyId
WHERE NOT EXISTS (SELECT 1 FROM dbo.Follow WITH (UPDLOCK, HOLDLOCK)
                  WHERE UserId = @userId AND StoryId = @storyId);
\end{minted}

\textit{Bỏ theo dõi truyện} (xoá dữ liệu):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
DELETE dbo.Follow WHERE UserId = @userId AND StoryId = @storyId;
\end{minted}

\textit{Tác giả đang theo dõi} (truy vấn kết nối ba bảng, phân trang):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT a.UserId AS AuthorId, a.PenName, u.AvatarUrl, a.IsVerified, a.FollowerCount,
       af.CreatedAt AS FollowedAt
FROM dbo.AuthorFollow af
JOIN dbo.Author a ON a.UserId = af.AuthorId
JOIN dbo.[User] u ON u.UserId = a.UserId
WHERE af.FollowerId = @userId
ORDER BY af.CreatedAt DESC, a.UserId DESC
OFFSET @offset ROWS FETCH NEXT @size ROWS ONLY;
\end{minted}

\textit{Lịch sử đọc} (truy vấn kết nối, kết nối ngoài để vẫn hiện truyện khi chương đã lưu không còn công khai):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT s.StoryId, s.Slug, s.Title, s.CoverUrl, s.ChapterCount, s.AuthorName,
       h.ScrollPercent, h.UpdatedAt AS ReadAt,
       ch.ChapterId AS LastChapterId, ch.ChapterNo AS LastChapterNo, ch.Title AS LastChapterTitle
FROM dbo.vw_PublicStory s
JOIN dbo.ReadingHistory h ON h.StoryId = s.StoryId AND h.UserId = @userId
LEFT JOIN dbo.Chapter ch  ON ch.ChapterId = h.LastChapterId
                         AND ch.Status = 'PUBLISHED' AND ch.IsDeleted = 0
ORDER BY h.UpdatedAt DESC, s.StoryId DESC
OFFSET @offset ROWS FETCH NEXT @size ROWS ONLY;
\end{minted}

\textit{Thêm truyện vào cuối tủ} (thêm dữ liệu từ truy vấn con có hàm tổng hợp; với tủ rỗng, \texttt{MAX} trả về \texttt{NULL} nên số thứ tự bắt đầu từ 1):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
INSERT dbo.ReadingListItem (ListId, StoryId, OrderNo, Note)
SELECT @listId, @storyId, ISNULL(MAX(i.OrderNo), 0) + 1, @note
FROM dbo.ReadingListItem i
WHERE i.ListId = @listId;

UPDATE dbo.ReadingList
SET ItemCount = ItemCount + 1, UpdatedAt = SYSUTCDATETIME()
WHERE ListId = @listId;
\end{minted}

\textit{Sắp xếp lại tủ} (cập nhật toàn bộ thứ tự mới trong một câu lệnh, dùng bảng giá trị \texttt{VALUES}). Ví dụ với ba truyện theo thứ tự mới 7, 3, 12:
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
UPDATE i
SET OrderNo = v.OrderNo
FROM dbo.ReadingListItem i
JOIN (VALUES (7, 1), (3, 2), (12, 3)) AS v (StoryId, OrderNo)
  ON v.StoryId = i.StoryId
WHERE i.ListId = @listId;
\end{minted}

\textit{Lưu đánh giá} (thêm mới hoặc cập nhật trong một câu lệnh):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
MERGE dbo.Rating WITH (HOLDLOCK) AS t
USING (SELECT @userId AS UserId, @storyId AS StoryId,
              CAST(@score AS TINYINT) AS Score, @review AS Review) AS src
   ON t.UserId = src.UserId AND t.StoryId = src.StoryId
WHEN MATCHED THEN
    UPDATE SET Score = src.Score, Review = src.Review, UpdatedAt = SYSUTCDATETIME()
WHEN NOT MATCHED THEN
    INSERT (UserId, StoryId, Score, Review)
    VALUES (src.UserId, src.StoryId, src.Score, src.Review);
\end{minted}

\textit{Trigger \texttt{trg\_Rating} tính lại điểm trung bình} sau mỗi lần thêm, sửa, xoá đánh giá (cập nhật có kết nối, truy vấn con và hàm tổng hợp; xử lý được nhiều dòng thay đổi cùng lúc):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
UPDATE s
SET RatingAvg   = COALESCE(x.AvgScore, 0),
    RatingCount = COALESCE(x.RatingCount, 0)
FROM dbo.Story s
JOIN (SELECT StoryId FROM inserted UNION SELECT StoryId FROM deleted) t
  ON t.StoryId = s.StoryId
OUTER APPLY (SELECT CAST(AVG(CAST(r.Score AS DECIMAL(5,2))) AS DECIMAL(3,2)) AS AvgScore,
                    COUNT(*) AS RatingCount
             FROM dbo.Rating r
             WHERE r.StoryId = s.StoryId) x;
\end{minted}

\textit{Phân bố điểm đánh giá của một truyện} (hàm tổng hợp với \texttt{GROUP BY}):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT r.Score, COUNT(*) AS Total
FROM dbo.Rating r
WHERE r.StoryId = @storyId
GROUP BY r.Score;
\end{minted}

\subsubsection{Module Yêu cầu tác giả}
\label{subsubsec:feature-author-request}

\paragraph{Mô tả chức năng.} Module hiện thực nhóm yêu cầu FR-7 và phần duyệt tác giả của FR-11.3:
\begin{itemize}
    \item \textbf{Gửi yêu cầu:} người đọc đang hoạt động gửi bút danh đề xuất và lời giới thiệu; yêu cầu mới có trạng thái \texttt{PENDING}. Mỗi người chỉ có tối đa một yêu cầu đang chờ, và bút danh không được trùng với tác giả đã có. Tác giả và quản trị viên không được gửi yêu cầu.
    \item \textbf{Theo dõi và huỷ yêu cầu:} người gửi xem lại toàn bộ yêu cầu của mình kèm trạng thái, ghi chú phản hồi và thời điểm xét duyệt; yêu cầu đang chờ có thể được huỷ (trạng thái \texttt{CANCELLED}).
    \item \textbf{Hàng đợi duyệt:} quản trị viên xem danh sách yêu cầu theo trạng thái (mặc định là đang chờ, cũ nhất trước), nhận biết người từng bị từ chối, và xem chi tiết từng yêu cầu kèm thông tin người gửi.
    \item \textbf{Duyệt và từ chối:} khi duyệt, trạng thái yêu cầu, vai trò tài khoản và hồ sơ tác giả được thay đổi trong cùng một giao dịch; khi từ chối, quản trị viên có thể kèm ghi chú phản hồi. Vai trò mới có hiệu lực ngay ở yêu cầu tiếp theo của người dùng, không cần đăng nhập lại.
\end{itemize}
Điểm quan trọng nhất của module là bảo đảm \textbf{mỗi yêu cầu chỉ được xử lý một lần}, kể cả khi hai quản trị viên bấm duyệt cùng lúc. Ràng buộc này được hiện thực bằng khoá dòng trong thủ tục lưu trữ trên SQL Server; ngoài ra, hệ thống có thể bật thêm khoá phân tán trên Redis để so sánh hai cơ chế kiểm soát đồng thời. \textit{Gửi yêu cầu} (thêm dữ liệu; trạng thái và thời điểm tạo lấy từ giá trị mặc định của cột):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
INSERT dbo.AuthorRequest (UserId, ProposedPenName, Introduction)
VALUES (@userId, @proposedPenName, @introduction);
\end{minted}

\textit{Kiểm tra yêu cầu đang chờ và bút danh} (truy vấn con \texttt{EXISTS}):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT CASE WHEN EXISTS (SELECT 1 FROM dbo.AuthorRequest
                         WHERE UserId = @userId AND Status = 'PENDING')
            THEN 1 ELSE 0 END AS HasPending;

SELECT CASE WHEN EXISTS (SELECT 1 FROM dbo.Author WHERE PenName = @penName)
            THEN 1 ELSE 0 END AS PenNameTaken;
\end{minted}

\textit{Yêu cầu của tôi} (truy vấn một điều kiện):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT RequestId, ProposedPenName, Introduction, Status, ReviewNote, CreatedAt, ReviewedAt
FROM dbo.AuthorRequest
WHERE UserId = @userId
ORDER BY CreatedAt DESC, RequestId DESC;
\end{minted}

\textit{Huỷ yêu cầu} (cập nhật nhiều điều kiện; điều kiện \texttt{Status = 'PENDING'} trong \texttt{WHERE} bảo đảm chỉ huỷ được yêu cầu còn đang chờ):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
UPDATE dbo.AuthorRequest
SET Status = 'CANCELLED'
WHERE RequestId = @requestId AND UserId = @userId AND Status = 'PENDING';
\end{minted}

\textit{Hàng đợi duyệt} (truy vấn kết nối và truy vấn con tương quan, đánh dấu người từng bị từ chối ở yêu cầu khác):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT r.RequestId, r.ProposedPenName, r.Status, r.CreatedAt, r.ReviewedAt,
       u.UserId, u.Username, u.AvatarUrl,
       CASE WHEN EXISTS (
           SELECT 1 FROM dbo.AuthorRequest r2
           WHERE r2.UserId = r.UserId AND r2.Status = 'REJECTED'
             AND r2.RequestId <> r.RequestId)
       THEN 1 ELSE 0 END AS PreviouslyRejected
FROM dbo.AuthorRequest r
JOIN dbo.[User] u ON u.UserId = r.UserId
WHERE r.Status = 'PENDING'
ORDER BY r.CreatedAt ASC, r.RequestId ASC
OFFSET @offset ROWS FETCH NEXT @size ROWS ONLY;
\end{minted}

\textit{Số yêu cầu khác của cùng người gửi theo trạng thái} (hàm tổng hợp với \texttt{GROUP BY} và truy vấn con):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT o.Status, COUNT(*) AS Total
FROM dbo.AuthorRequest o
WHERE o.UserId = (SELECT r.UserId FROM dbo.AuthorRequest r WHERE r.RequestId = @requestId)
  AND o.RequestId <> @requestId
GROUP BY o.Status;
\end{minted}

\textit{Thủ tục duyệt yêu cầu \texttt{sp\_ApproveAuthorRequest}} (giao dịch gồm khoá dòng, các kiểm tra và ba thao tác ghi). Gợi ý \texttt{UPDLOCK, HOLDLOCK} khiến phiên duyệt thứ hai phải chờ phiên thứ nhất kết thúc rồi mới đọc được trạng thái:
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
BEGIN TRANSACTION;

SELECT @UserId = UserId, @PenName = ProposedPenName, @Status = Status
FROM dbo.AuthorRequest WITH (UPDLOCK, HOLDLOCK)
WHERE RequestId = @RequestId;

IF @@ROWCOUNT = 0
    THROW 50004, N'Author request not found.', 1;
IF NOT EXISTS (SELECT 1 FROM dbo.[User]
               WHERE UserId = @AdminId AND Role = 'ADMIN' AND Status = 'ACTIVE')
    THROW 50002, N'Only an active ADMIN can review author requests.', 1;
IF @Status <> 'PENDING'
    THROW 50001, N'Author request was already processed.', 1;
IF NOT EXISTS (SELECT 1 FROM dbo.[User] WITH (UPDLOCK)
               WHERE UserId = @UserId AND Role = 'READER' AND Status = 'ACTIVE')
    THROW 50002, N'The applicant is not an active READER.', 1;
IF EXISTS (SELECT 1 FROM dbo.Author WITH (UPDLOCK, HOLDLOCK) WHERE PenName = @PenName)
    THROW 50003, N'Pen name is already in use.', 1;

UPDATE dbo.AuthorRequest
SET Status = 'APPROVED', ReviewedBy = @AdminId, ReviewedAt = SYSUTCDATETIME()
WHERE RequestId = @RequestId;

UPDATE dbo.[User]
SET Role = 'AUTHOR', UpdatedAt = SYSUTCDATETIME()
WHERE UserId = @UserId;

INSERT dbo.Author (UserId, PenName, ApprovedRequestId)
VALUES (@UserId, @PenName, @RequestId);

COMMIT TRANSACTION;
\end{minted}
Thủ tục được bọc trong \texttt{BEGIN TRY \ldots{} END TRY} với \texttt{SET XACT\_ABORT ON}; khi có lỗi, khối \texttt{CATCH} huỷ giao dịch và ném lại lỗi để backend chuyển thành mã HTTP tương ứng. Thủ tục từ chối \texttt{sp\_RejectAuthorRequest} có cấu trúc tương tự, chỉ cập nhật trạng thái \texttt{REJECTED} và ghi chú phản hồi.

\subsubsection{Module Studio sáng tác}
\label{subsubsec:feature-studio}

\paragraph{Mô tả chức năng.} Module là khu vực làm việc của tác giả, hiện thực các nhóm yêu cầu FR-8, FR-9 và FR-10. Mọi thao tác chỉ áp dụng trên truyện của chính tác giả đang đăng nhập; truyện hoặc chương của người khác đều được trả về như không tồn tại.
\begin{itemize}
    \item \textbf{Quản lý truyện:} danh sách truyện của tôi lọc theo trạng thái; tạo truyện mới ở trạng thái bản nháp với slug sinh tự động từ tên truyện; sửa thông tin, thay thể loại (1--3, đúng một thể loại chính) và thẻ (tối đa 10); công khai truyện khi đã có ít nhất một chương đăng, chuyển về bản nháp hoặc xoá mềm truyện; cập nhật hồ sơ tác giả.
    \item \textbf{Viết chương:} tạo, đổi tên, xoá tập (chỉ khi tập không còn chương); tạo chương với số thứ tự tự tăng; tự động lưu bản nháp lên Redis trong lúc soạn; lưu chương; đăng ngay hoặc hẹn giờ đăng; xoá chương.
    \item \textbf{Phiên bản chương:} mỗi lần nội dung chương thay đổi, bản cũ được lưu thành một phiên bản; tác giả xem tối đa 10 phiên bản gần nhất và khôi phục chương về một phiên bản đã chọn.
    \item \textbf{Thống kê tác giả:} số liệu tổng quan, biểu đồ lượt xem theo ngày hoặc tháng (toàn bộ hoặc từng truyện), thống kê từng truyện kèm lượt xem 30 ngày và đánh dấu truyện có điểm cao hơn mức trung bình của chính tác giả, thông tin người theo dõi.
\end{itemize}
Việc sửa truyện và sửa chương dùng \textbf{khoá lạc quan} theo cột \texttt{UpdatedAt}: lệnh \texttt{UPDATE} chỉ áp dụng khi \texttt{UpdatedAt} còn bằng giá trị mà giao diện đã đọc, nếu không thì báo lỗi \texttt{409} để tránh ghi đè chỉnh sửa ở một tab khác. Đăng chương được giới hạn 10 lần mỗi phút, tạo truyện 10 lần mỗi giờ và lưu bản nháp 30 lần mỗi phút trên một tài khoản.

\paragraph{Các câu truy vấn SQL.}

\textit{Truyện của tôi theo trạng thái} (truy vấn nhiều điều kiện, phân trang):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT s.StoryId, s.Slug, s.Title, s.CoverUrl, s.PublishStatus, s.ChapterCount,
       s.ViewCount, s.FollowCount, s.RatingAvg, s.UpdatedAt
FROM dbo.Story s
WHERE s.AuthorId = @authorId AND s.IsDeleted = 0 AND s.PublishStatus = @status
ORDER BY s.UpdatedAt DESC, s.StoryId DESC
OFFSET @offset ROWS FETCH NEXT @size ROWS ONLY;
\end{minted}

\textit{Tạo truyện và gán thể loại} (thêm dữ liệu; các cột trạng thái, bộ đếm và thời gian lấy giá trị mặc định). Ví dụ với hai thể loại, thể loại 3 là thể loại chính:
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
INSERT dbo.Story (AuthorId, Title, Slug, Description, CoverUrl, AgeRating, CompletionStatus)
VALUES (@authorId, @title, @slug, @description, @coverUrl, @ageRating, @completionStatus);

INSERT dbo.StoryGenre (StoryId, GenreId, IsPrimary)
VALUES (@storyId, 3, 1), (@storyId, 7, 0);
\end{minted}

\textit{Sửa thông tin truyện với khoá lạc quan} (cập nhật nhiều điều kiện; 0 dòng bị ảnh hưởng nghĩa là dữ liệu đã bị sửa ở nơi khác):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
UPDATE dbo.Story
SET Title = @title, Description = @description, CompletionStatus = @completionStatus,
    UpdatedAt = SYSUTCDATETIME()
WHERE StoryId = @storyId AND AuthorId = @authorId AND IsDeleted = 0
  AND UpdatedAt = @version;
\end{minted}

\textit{Công khai truyện} (cập nhật có điều kiện tiên quyết nằm trong \texttt{WHERE}, nên kiểm tra và đổi trạng thái diễn ra trong một câu lệnh):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
UPDATE dbo.Story
SET PublishStatus = 'PUBLISHED',
    PublishedAt   = COALESCE(PublishedAt, SYSUTCDATETIME()),
    UpdatedAt     = SYSUTCDATETIME()
WHERE StoryId = @storyId AND AuthorId = @authorId AND IsDeleted = 0
  AND PublishStatus = 'DRAFT' AND ChapterCount > 0;
\end{minted}

\textit{Tạo chương với số thứ tự kế tiếp} (thêm dữ liệu từ truy vấn con có hàm tổng hợp; \texttt{UPDLOCK, HOLDLOCK} khoá khoảng khoá của truyện, nên hai lệnh tạo đồng thời nhận hai số khác nhau):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
INSERT dbo.Chapter (StoryId, VolumeId, ChapterNo, Title, Content, WordCount)
SELECT @storyId, @volumeId, ISNULL(MAX(x.ChapterNo), 0) + 1, @title, @content, @wordCount
FROM dbo.Chapter x WITH (UPDLOCK, HOLDLOCK)
WHERE x.StoryId = @storyId;
\end{minted}

\textit{Lưu chương} (cập nhật với khoá lạc quan; mã tác giả được đặt vào \texttt{SESSION\_CONTEXT} trước, trên cùng kết nối):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
EXEC sys.sp_set_session_context @key = N'user_id', @value = @authorId;

UPDATE dbo.Chapter
SET Title = @title, Content = @content, WordCount = @wordCount,
    VolumeId = @volumeId, UpdatedAt = SYSUTCDATETIME()
WHERE ChapterId = @chapterId AND IsDeleted = 0 AND UpdatedAt = @version;
\end{minted}

\textit{Trigger \texttt{trg\_Chapter\_Revision} lưu phiên bản cũ} (thêm dữ liệu từ bảng \texttt{deleted} với truy vấn con, sau đó xoá các phiên bản cũ hơn 10 bản gần nhất bằng hàm cửa sổ \texttt{ROW\_NUMBER}):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
INSERT dbo.ChapterRevision (ChapterId, RevisionNo, Title, Content, WordCount, EditedBy)
SELECT d.ChapterId, COALESCE(m.MaxNo, 0) + 1, d.Title, d.Content, d.WordCount,
       CAST(SESSION_CONTEXT(N'user_id') AS INT)
FROM deleted d
JOIN inserted i ON i.ChapterId = d.ChapterId
OUTER APPLY (SELECT MAX(r.RevisionNo) AS MaxNo
             FROM dbo.ChapterRevision r WITH (UPDLOCK, HOLDLOCK)
             WHERE r.ChapterId = d.ChapterId) m
WHERE i.Title   COLLATE Latin1_General_100_BIN2 <> d.Title   COLLATE Latin1_General_100_BIN2
   OR i.Content COLLATE Latin1_General_100_BIN2 <> d.Content COLLATE Latin1_General_100_BIN2;

WITH ranked AS (
    SELECT RevisionId,
           ROW_NUMBER() OVER (PARTITION BY ChapterId ORDER BY RevisionNo DESC) AS rn
    FROM dbo.ChapterRevision
    WHERE ChapterId IN (SELECT ChapterId FROM inserted)
)
DELETE FROM ranked WHERE rn > 10;
\end{minted}

\textit{Tính lại thống kê truyện} trong thủ tục \texttt{sp\_RecalcStoryStats}, được gọi khi đăng, sửa hoặc xoá chương đã đăng (cập nhật với \texttt{CROSS APPLY} và hàm tổng hợp):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
UPDATE s
SET ChapterCount  = x.ChapterCount,
    WordCount     = x.WordCount,
    LastChapterAt = x.LastChapterAt,
    UpdatedAt     = SYSUTCDATETIME()
FROM dbo.Story s
CROSS APPLY (SELECT COUNT(*)                                     AS ChapterCount,
                    COALESCE(SUM(CAST(c.WordCount AS BIGINT)), 0) AS WordCount,
                    MAX(c.PublishedAt)                            AS LastChapterAt
             FROM dbo.Chapter c
             WHERE c.StoryId = s.StoryId AND c.Status = 'PUBLISHED' AND c.IsDeleted = 0) x
WHERE s.StoryId = @storyId;
\end{minted}

\textit{Chọn các chương hẹn giờ đã đến hạn} trong thủ tục \texttt{sp\_PublishScheduled}; \texttt{READPAST} bỏ qua các dòng đang bị khoá:
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
INSERT @Due (ChapterId, StoryId)
SELECT c.ChapterId, c.StoryId
FROM dbo.Chapter c WITH (UPDLOCK, READPAST, ROWLOCK)
WHERE c.Status = 'SCHEDULED' AND c.IsDeleted = 0 AND c.ScheduledAt <= @Now;

UPDATE c
SET Status = 'PUBLISHED', PublishedAt = @Now, ScheduledAt = NULL, UpdatedAt = @Now
FROM dbo.Chapter c
JOIN @Due d ON d.ChapterId = c.ChapterId;
\end{minted}

\textit{Danh sách phiên bản của chương} (truy vấn kết nối):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT r.RevisionNo, r.Title, r.WordCount, a.PenName AS EditedBy, r.CreatedAt
FROM dbo.ChapterRevision r
JOIN dbo.Author a ON a.UserId = r.EditedBy
WHERE r.ChapterId = @chapterId
ORDER BY r.RevisionNo DESC;
\end{minted}

\textit{Lượt xem theo tháng} (truy vấn kết nối, hàm tổng hợp và \texttt{GROUP BY}; khi xem theo ngày, biểu thức nhóm là \texttt{d.ViewDate}):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT DATEFROMPARTS(YEAR(d.ViewDate), MONTH(d.ViewDate), 1) AS Period,
       SUM(CAST(d.Views AS BIGINT)) AS Views
FROM dbo.StoryViewDaily d
JOIN dbo.Story s ON s.StoryId = d.StoryId
WHERE s.AuthorId = @authorId AND s.IsDeleted = 0
  AND d.ViewDate BETWEEN @from AND @to
GROUP BY DATEFROMPARTS(YEAR(d.ViewDate), MONTH(d.ViewDate), 1)
ORDER BY Period;
\end{minted}

\textit{Thống kê từng truyện} (truy vấn con tương quan so sánh với điểm trung bình các truyện của chính tác giả, và \texttt{OUTER APPLY} tính lượt xem 30 ngày):
\begin{minted}[fontsize=\small,frame=single,breaklines]{sql}
SELECT s.StoryId, s.Title, s.ViewCount, s.RatingAvg, s.RatingCount,
       COALESCE(v.Views, 0) AS Views30d,
       CASE WHEN s.RatingCount > 0
                 AND s.RatingAvg > (SELECT AVG(s2.RatingAvg)
                                    FROM dbo.Story s2
                                    WHERE s2.AuthorId = s.AuthorId AND s2.IsDeleted = 0
                                      AND s2.RatingCount > 0)
            THEN 1 ELSE 0 END AS AboveMyAverage
FROM dbo.Story s
OUTER APPLY (SELECT SUM(CAST(d.Views AS BIGINT)) AS Views
             FROM dbo.StoryViewDaily d
             WHERE d.StoryId = s.StoryId AND d.ViewDate >= @since) v
WHERE s.AuthorId = @authorId AND s.IsDeleted = 0
ORDER BY s.ViewCount DESC, s.StoryId;
\end{minted}
